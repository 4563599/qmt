{20阴7选股公式}

{定义：强阴线 = 阴线且最低价大于5日均价}
{定义：弱阴线 = 阴线且最低价小于等于5日均价}

{计算5日均价}
MA5:=MA(C,5);

{判断是否为阴线}
ISYIN:=C<O;

{判断是否为强阴线：阴线且最低价大于5日均价}
QYIN:=ISYIN AND L>MA5;

{判断是否为弱阴线：阴线且最低价小于等于5日均价}
RYIN:=ISYIN AND L<=MA5;

{判断最低价是否大于5日均价}
LGTMA5:=L>MA5;

{统计最近20个交易日的数据}
{最近20个交易日阴线次数}
YINC20:=COUNT(ISYIN,20);

{最近20个交易日弱阴线次数}
RYINC20:=COUNT(RYIN,20);

{最近20个交易日最低价大于5日均价次数}
LGC20:=COUNT(LGTMA5,20);

{选股条件}
COND1:=RYINC20<=3;      {弱阴线次数<=3}
COND2:=YINC20<=7;       {阴线次数<=7}
COND3:=LGC20>=7;        {最低价大于5日均价次数>=7}

{最终选股条件：三个条件同时满足}
COND1 AND COND2 AND COND3;
